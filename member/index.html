"use strict"; const express = require("express"); const crypto = require("crypto"); module.exports = function createMemberRegisterApiRoutes({ supabase, memberOtpSecret }) { if (!supabase) { throw new Error("supabase is required"); } if (!memberOtpSecret) { throw new Error("memberOtpSecret is required"); } const router = express.Router(); const REGISTRATION_TABLE = "02_member_registrations"; const MEMBER_TABLE = "03_members"; const TERMS_VERSION = "ADT-TERMS-1.0"; const PRIVACY_NOTICE_VERSION = "ADT-PRIVACY-1.0"; // ======================================================== // Helpers // ======================================================== function normalizeEmail(value) { return String(value || "") .trim() .toLowerCase(); } function normalizePhone(value) { return String(value || "") .replace(/\D/g, ""); } function normalizeUsername(value) { return String(value || "") .trim(); } function isValidEmail(value) { return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test( value ); } function isValidThaiPhone(value) { return /^0\d{9}$/.test(value); } function isValidUsername(value) { return /^[A-Za-z0-9._-]{4,40}$/.test( value ); } function isValidPassword(value) { return ( typeof value === "string" && value.length >= 8 && value.length <= 200 ); } function hashOtp(otp) { return crypto .createHmac( "sha256", memberOtpSecret ) .update(String(otp)) .digest("hex"); } function sendError( res, status, code, message ) { return res.status(status).json({ success: false, code, message }); } function isUniqueViolation(error) { return ( error?.code === "23505" || String(error?.message || "") .toLowerCase() .includes("duplicate key") ); } // ======================================================== // Verify LINE OTP // Save registration data // Reserve member code // // POST /api/member/register/verify-line-otp // ======================================================== router.post( "/register/verify-line-otp", async (req, res) => { try { const fullName = String( req.body?.full_name || "" ).trim(); const currentAddress = String( req.body?.current_address || "" ).trim(); const email = normalizeEmail( req.body?.email ); const phone = normalizePhone( req.body?.phone ); const otp = String( req.body?.otp || "" ).trim(); // ================================================== // Validate form data // ================================================== if (fullName.length < 2) { return sendError( res, 400, "INVALID_FULL_NAME", "กรุณากรอกชื่อ–นามสกุล" ); } if ( currentAddress.length < 10 ) { return sendError( res, 400, "INVALID_ADDRESS", "กรุณากรอกที่อยู่ติดต่อให้ครบถ้วน" ); } if (!isValidEmail(email)) { return sendError( res, 400, "INVALID_EMAIL", "รูปแบบอีเมลไม่ถูกต้อง" ); } if ( !isValidThaiPhone(phone) ) { return sendError( res, 400, "INVALID_PHONE", "เบอร์โทรศัพท์ต้องมี 10 หลัก" ); } if ( !/^\d{6}$/.test(otp) ) { return sendError( res, 400, "INVALID_OTP_FORMAT", "กรุณากรอกรหัส OTP ตัวเลข 6 หลัก" ); } const otpHash = hashOtp(otp); const nowIso = new Date().toISOString(); // ================================================== // Check existing member by email or phone // ================================================== const { data: existingMembers, error: existingMemberError } = await supabase .from(MEMBER_TABLE) .select( "id, member_code, email, phone" ) .or( `email.eq.${email},phone.eq.${phone}` ) .limit(2); if (existingMemberError) { throw existingMemberError; } if ( existingMembers?.length ) { const existingByEmail = existingMembers.some( member => String( member.email || "" ).toLowerCase() === email ); return sendError( res, 409, existingByEmail ? "EMAIL_ALREADY_MEMBER" : "PHONE_ALREADY_MEMBER", existingByEmail ? "อีเมลนี้เป็นสมาชิก ADT อยู่แล้ว" : "เบอร์โทรศัพท์นี้เป็นสมาชิก ADT อยู่แล้ว" ); } // ================================================== // Find active OTP created by old LINE API // ================================================== const { data: otpRows, error: otpLookupError } = await supabase .from( REGISTRATION_TABLE ) .select(` id, line_user_id, line_display_name, line_picture_url, otp_hash, otp_expires_at, otp_attempts, status `) .eq( "otp_hash", otpHash ) .eq( "status", "OTP_SENT" ) .gt( "otp_expires_at", nowIso ) .order( "created_at", { ascending: false } ) .limit(2); if (otpLookupError) { throw otpLookupError; } if ( !otpRows || otpRows.length === 0 ) { return sendError( res, 400, "OTP_INVALID_OR_EXPIRED", "รหัส OTP ไม่ถูกต้องหรือหมดอายุแล้ว" ); } if ( otpRows.length > 1 ) { console.error( "Duplicate active OTP hash detected:", otpHash ); return sendError( res, 409, "OTP_CONFLICT", "พบรหัส OTP ซ้ำในระบบ กรุณาขอรหัสใหม่" ); } const registration = otpRows[0]; if ( !registration.line_user_id ) { return sendError( res, 409, "LINE_ID_NOT_FOUND", "ไม่พบข้อมูลบัญชี LINE ของรหัส OTP นี้" ); } // ================================================== // Check existing member by LINE ID // ================================================== const { data: existingLineMember, error: existingLineError } = await supabase .from(MEMBER_TABLE) .select( "id, member_code" ) .eq( "line_user_id", registration.line_user_id ) .maybeSingle(); if (existingLineError) { throw existingLineError; } if (existingLineMember) { return sendError( res, 409, "LINE_ALREADY_MEMBER", `บัญชี LINE นี้เป็นสมาชิกแล้ว หมายเลข ${existingLineMember.member_code}` ); } // ================================================== // Verify OTP once // Save profile data // ================================================== const { data: verifiedRegistration, error: updateError } = await supabase .from( REGISTRATION_TABLE ) .update({ full_name: fullName, current_address: currentAddress, email, phone, otp_verified_at: nowIso, otp_hash: null, otp_expires_at: null, otp_attempts: 0, status: "BOT_VERIFIED" }) .eq( "id", registration.id ) .eq( "status", "OTP_SENT" ) .eq( "otp_hash", otpHash ) .gt( "otp_expires_at", nowIso ) .select(` id, line_user_id, status `) .maybeSingle(); if (updateError) { throw updateError; } if ( !verifiedRegistration ) { return sendError( res, 409, "OTP_ALREADY_USED", "รหัส OTP นี้ถูกใช้ไปแล้วหรือหมดอายุแล้ว" ); } // ================================================== // Reserve global ADT member code // ================================================== const { data: memberCode, error: memberCodeError } = await supabase.rpc( "reserve_adt_member_code", { p_registration_id: verifiedRegistration.id } ); if (memberCodeError) { console.error( "Reserve member code error:", memberCodeError ); return sendError( res, 500, "MEMBER_CODE_FAILED", "ยืนยัน OTP สำเร็จ แต่ไม่สามารถออกหมายเลขสมาชิกได้" ); } return res .status(200) .json({ success: true, registration_id: verifiedRegistration.id, member_code: memberCode, next_step: "CREATE_ACCOUNT" }); } catch (error) { console.error( "Verify LINE OTP error:", error ); return sendError( res, 500, "VERIFY_LINE_OTP_FAILED", "ตรวจสอบ OTP และบันทึกข้อมูลสมาชิกไม่สำเร็จ" ); } } ); // ======================================================== // Complete registration // Create Supabase Auth user // Create member record // // POST /api/member/register/complete // ======================================================== router.post( "/register/complete", async (req, res) => { let createdAuthUserId = null; try { const registrationId = String( req.body ?.registration_id || "" ).trim(); const requestedMemberCode = String( req.body?.member_code || "" ).trim(); const username = normalizeUsername( req.body?.username ); const password = String( req.body?.password || "" ); const agreeTerms = req.body ?.agree_terms === true; const privacyAgreed = req.body ?.privacy_agreed === undefined ? agreeTerms : req.body ?.privacy_agreed === true; const marketingConsent = req.body ?.marketing_consent === true; // ================================================== // Validate request // ================================================== if (!registrationId) { return sendError( res, 400, "REGISTRATION_ID_REQUIRED", "ไม่พบรายการสมัครสมาชิก" ); } if ( !requestedMemberCode ) { return sendError( res, 400, "MEMBER_CODE_REQUIRED", "ไม่พบหมายเลขสมาชิก" ); } if ( !/^ADT-M\d{6}$/.test( requestedMemberCode ) ) { return sendError( res, 400, "INVALID_MEMBER_CODE", "รูปแบบหมายเลขสมาชิกไม่ถูกต้อง" ); } if ( !isValidUsername( username ) ) { return sendError( res, 400, "INVALID_USERNAME", "Username ต้องมี 4–40 ตัวอักษร และใช้ได้เฉพาะภาษาอังกฤษ ตัวเลข จุด ขีดกลาง หรือขีดล่าง" ); } if ( !isValidPassword( password ) ) { return sendError( res, 400, "INVALID_PASSWORD", "Password ต้องมีอย่างน้อย 8 ตัวอักษร" ); } if ( !agreeTerms || !privacyAgreed ) { return sendError( res, 400, "CONSENT_REQUIRED", "กรุณาอ่านและยอมรับข้อกำหนดการใช้งานและประกาศความเป็นส่วนตัว" ); } // ================================================== // Read verified registration // ================================================== const { data: registration, error: registrationError } = await supabase .from( REGISTRATION_TABLE ) .select(` id, full_name, current_address, email, phone, line_user_id, line_display_name, line_picture_url, reserved_member_no, reserved_member_code, status, completed_at `) .eq( "id", registrationId ) .maybeSingle(); if (registrationError) { throw registrationError; } if (!registration) { return sendError( res, 404, "REGISTRATION_NOT_FOUND", "ไม่พบรายการสมัครสมาชิก" ); } if ( registration.status === "COMPLETED" ) { return sendError( res, 409, "REGISTRATION_COMPLETED", "รายการสมัครนี้สร้างบัญชีเรียบร้อยแล้ว" ); } if ( registration.status !== "ACCOUNT_PENDING" ) { return sendError( res, 409, "REGISTRATION_NOT_READY", "รายการสมัครยังไม่พร้อมสร้างบัญชี" ); } if ( registration .reserved_member_code !== requestedMemberCode ) { return sendError( res, 409, "MEMBER_CODE_MISMATCH", "หมายเลขสมาชิกไม่ตรงกับรายการสมัคร" ); } if ( !registration .reserved_member_no || !registration .reserved_member_code ) { return sendError( res, 409, "MEMBER_CODE_NOT_RESERVED", "รายการสมัครยังไม่มีหมายเลขสมาชิก" ); } if ( !registration.full_name || !registration .current_address || !registration.email || !registration.phone || !registration .line_user_id ) { return sendError( res, 409, "REGISTRATION_INCOMPLETE", "ข้อมูลรายการสมัครไม่ครบถ้วน" ); } const email = normalizeEmail( registration.email ); const phone = normalizePhone( registration.phone ); if (!isValidEmail(email)) { return sendError( res, 409, "REGISTRATION_EMAIL_INVALID", "อีเมลในรายการสมัครไม่ถูกต้อง" ); } if ( !isValidThaiPhone(phone) ) { return sendError( res, 409, "REGISTRATION_PHONE_INVALID", "เบอร์โทรศัพท์ในรายการสมัครไม่ถูกต้อง" ); } // ================================================== // Check duplicate username // ================================================== const { data: existingUsername, error: existingUsernameError } = await supabase .from(MEMBER_TABLE) .select( "id, username" ) .ilike( "username", username ) .limit(1) .maybeSingle(); if ( existingUsernameError ) { throw existingUsernameError; } if (existingUsername) { return sendError( res, 409, "USERNAME_ALREADY_EXISTS", "Username นี้ถูกใช้งานแล้ว กรุณาเลือกชื่ออื่น" ); } // ================================================== // Check duplicate email, phone, LINE and member code // ================================================== const { data: existingMemberRows, error: existingMemberRowsError } = await supabase .from(MEMBER_TABLE) .select(` id, member_code, email, phone, line_user_id `) .or( [ `email.eq.${email}`, `phone.eq.${phone}`, `line_user_id.eq.${registration.line_user_id}`, `member_code.eq.${registration.reserved_member_code}` ].join(",") ) .limit(4); if ( existingMemberRowsError ) { throw existingMemberRowsError; } if ( existingMemberRows ?.length ) { const duplicate = existingMemberRows[0]; if ( String( duplicate.email || "" ).toLowerCase() === email ) { return sendError( res, 409, "EMAIL_ALREADY_MEMBER", "อีเมลนี้เป็นสมาชิก ADT อยู่แล้ว" ); } if ( duplicate.phone === phone ) { return sendError( res, 409, "PHONE_ALREADY_MEMBER", "เบอร์โทรศัพท์นี้เป็นสมาชิก ADT อยู่แล้ว" ); } if ( duplicate .line_user_id === registration .line_user_id ) { return sendError( res, 409, "LINE_ALREADY_MEMBER", "บัญชี LINE นี้เป็นสมาชิก ADT อยู่แล้ว" ); } return sendError( res, 409, "MEMBER_CODE_ALREADY_EXISTS", "หมายเลขสมาชิกนี้ถูกใช้งานแล้ว" ); } // ================================================== // Create Supabase Auth user // ================================================== const { data: authResult, error: authError } = await supabase .auth .admin .createUser({ email, password, email_confirm: true, user_metadata: { username, full_name: registration .full_name, member_code: registration .reserved_member_code }, app_metadata: { role: "member" } }); if (authError) { console.error( "Create auth user error:", authError ); const authMessage = String( authError.message || "" ).toLowerCase(); if ( authMessage.includes( "already" ) || authMessage.includes( "registered" ) || authMessage.includes( "exists" ) ) { return sendError( res, 409, "AUTH_EMAIL_ALREADY_EXISTS", "อีเมลนี้มีบัญชีในระบบอยู่แล้ว" ); } return sendError( res, 500, "AUTH_USER_CREATE_FAILED", "ไม่สามารถสร้างบัญชีเข้าสู่ระบบได้" ); } createdAuthUserId = authResult?.user?.id; if ( !createdAuthUserId ) { return sendError( res, 500, "AUTH_USER_ID_MISSING", "สร้างบัญชีแล้วแต่ไม่พบรหัสผู้ใช้งาน" ); } const nowIso = new Date() .toISOString(); // ================================================== // Insert member record // ================================================== const { data: member, error: memberInsertError } = await supabase .from(MEMBER_TABLE) .insert({ auth_user_id: createdAuthUserId, member_no: registration .reserved_member_no, member_code: registration .reserved_member_code, username, full_name: registration .full_name, current_address: registration .current_address, email, phone, line_user_id: registration .line_user_id, line_display_name: registration .line_display_name, line_picture_url: registration .line_picture_url, display_name: registration .full_name, avatar_url: registration .line_picture_url, status: "ACTIVE", is_verified: true, is_active: true, profile_completion: 25, privacy_notice_version: PRIVACY_NOTICE_VERSION, terms_version: TERMS_VERSION, privacy_agreed: true, privacy_agreed_at: nowIso, marketing_consent: marketingConsent, marketing_consent_at: marketingConsent ? nowIso : null, registration_id: registration.id }) .select(` id, member_no, member_code, username, full_name, email, status, created_at `) .single(); if (memberInsertError) { console.error( "Insert member error:", memberInsertError ); // Rollback Auth user if ( createdAuthUserId ) { const { error: rollbackAuthError } = await supabase .auth .admin .deleteUser( createdAuthUserId ); if ( rollbackAuthError ) { console.error( "Rollback auth user error:", rollbackAuthError ); } createdAuthUserId = null; } if ( isUniqueViolation( memberInsertError ) ) { return sendError( res, 409, "MEMBER_DUPLICATE", "Username อีเมล เบอร์โทร LINE หรือหมายเลขสมาชิกถูกใช้งานแล้ว" ); } return sendError( res, 500, "MEMBER_CREATE_FAILED", "ไม่สามารถบันทึกข้อมูลสมาชิกได้" ); } // ================================================== // Mark registration completed // ================================================== const { data: completedRegistration, error: registrationUpdateError } = await supabase .from( REGISTRATION_TABLE ) .update({ status: "COMPLETED", completed_at: nowIso }) .eq( "id", registration.id ) .eq( "status", "ACCOUNT_PENDING" ) .select( "id, status, completed_at" ) .maybeSingle(); if ( registrationUpdateError || !completedRegistration ) { console.error( "Complete registration update error:", registrationUpdateError ); // Member และ Auth ถูกสร้างแล้ว // ไม่ควรลบอัตโนมัติเพราะบัญชีใช้งานได้จริง // ส่งผลสำเร็จพร้อม warning ให้ตรวจสอบ Log ภายหลัง return res .status(200) .json({ success: true, warning: "MEMBER_CREATED_REGISTRATION_STATUS_PENDING", message: "สร้างบัญชีสมาชิกสำเร็จ แต่สถานะรายการสมัครยังอัปเดตไม่สมบูรณ์", member }); } return res .status(200) .json({ success: true, message: "สร้างบัญชีสมาชิกสำเร็จ", member }); } catch (error) { console.error( "Complete member registration error:", error ); // Rollback Auth user only when member was not created if (createdAuthUserId) { try { const { data: existingCreatedMember } = await supabase .from(MEMBER_TABLE) .select("id") .eq( "auth_user_id", createdAuthUserId ) .maybeSingle(); if ( !existingCreatedMember ) { const { error: rollbackError } = await supabase .auth .admin .deleteUser( createdAuthUserId ); if (rollbackError) { console.error( "Final auth rollback error:", rollbackError ); } } } catch ( rollbackCheckError ) { console.error( "Rollback check error:", rollbackCheckError ); } } return sendError( res, 500, "COMPLETE_REGISTRATION_FAILED", "ไม่สามารถสร้างบัญชีสมาชิกได้" ); } } ); // ======================================================== // Member Login // Find member by Username // Verify Password with Supabase Auth // // POST /api/member/login // ======================================================== router.post( "/login", async (req, res) => { try { const username = normalizeUsername( req.body?.username ); const password = String( req.body?.password || "" ); const remember = req.body?.remember === true; // ================================================== // Validate request // ================================================== if ( !isValidUsername( username ) ) { return sendError( res, 400, "INVALID_USERNAME", "กรุณากรอก Username ให้ถูกต้อง" ); } if ( !password || password.length < 8 || password.length > 200 ) { return sendError( res, 400, "INVALID_PASSWORD", "กรุณากรอก Password ให้ถูกต้อง" ); } // ================================================== // Find member in 03_members // Username is checked from the membership database // ================================================== const { data: member, error: memberLookupError } = await supabase .from(MEMBER_TABLE) .select(` id, auth_user_id, member_no, member_code, username, full_name, display_name, email, line_picture_url, avatar_url, status, is_verified, is_active, profile_completion `) .eq( "username", username ) .maybeSingle(); if (memberLookupError) { console.error( "Member login lookup error:", memberLookupError ); return sendError( res, 500, "MEMBER_LOOKUP_FAILED", "ไม่สามารถตรวจสอบข้อมูลสมาชิกได้" ); } if (!member) { return sendError( res, 401, "INVALID_CREDENTIALS", "Username หรือ Password ไม่ถูกต้อง" ); } // ================================================== // Check membership status before Auth login // ================================================== if ( member.status === "SUSPENDED" ) { return sendError( res, 403, "MEMBER_SUSPENDED", "บัญชีสมาชิกนี้ถูกระงับ กรุณาติดต่อทีมงาน ADT" ); } if ( member.status === "CLOSED" ) { return sendError( res, 403, "MEMBER_CLOSED", "บัญชีสมาชิกนี้ถูกปิดแล้ว" ); } if ( member.status !== "ACTIVE" || member.is_active !== true ) { return sendError( res, 403, "MEMBER_NOT_ACTIVE", "บัญชีสมาชิกยังไม่พร้อมใช้งาน" ); } if ( !member.auth_user_id || !member.email ) { return sendError( res, 409, "AUTH_ACCOUNT_NOT_LINKED", "บัญชีสมาชิกยังไม่ได้เชื่อมกับระบบเข้าสู่ระบบ" ); } // ================================================== // Verify Password with Supabase Auth // Password is never stored in 03_members // ================================================== const { data: authResult, error: authError } = await supabase .auth .signInWithPassword({ email: normalizeEmail( member.email ), password }); if ( authError || !authResult?.user || !authResult?.session ) { console.warn( "Member login rejected:", { username, reason: authError?.message || "No session returned" } ); return sendError( res, 401, "INVALID_CREDENTIALS", "Username หรือ Password ไม่ถูกต้อง" ); } // Prevent an Auth account from being used with another member row if ( authResult.user.id !== member.auth_user_id ) { console.error( "Member auth_user_id mismatch:", { member_id: member.id, member_auth_user_id: member.auth_user_id, authenticated_user_id: authResult.user.id } ); return sendError( res, 403, "AUTH_MEMBER_MISMATCH", "บัญชีเข้าสู่ระบบไม่ตรงกับข้อมูลสมาชิก" ); } const nowIso = new Date() .toISOString(); // ================================================== // Update last login time // Failure here must not block a valid login // ================================================== const { error: lastLoginError } = await supabase .from(MEMBER_TABLE) .update({ last_login_at: nowIso }) .eq( "id", member.id ); if (lastLoginError) { console.error( "Update member last_login_at error:", lastLoginError ); } // ================================================== // Success // The frontend stores tokens according to Remember Me // ================================================== return res .status(200) .json({ success: true, message: "เข้าสู่ระบบสำเร็จ", remember, session: { access_token: authResult .session .access_token, refresh_token: authResult .session .refresh_token, expires_at: authResult .session .expires_at, expires_in: authResult .session .expires_in, token_type: authResult .session .token_type }, member: { id: member.id, member_no: member.member_no, member_code: member.member_code, username: member.username, full_name: member.full_name, display_name: member.display_name || member.full_name, avatar_url: member.avatar_url || member.line_picture_url || null, status: member.status, is_verified: member.is_verified, profile_completion: member.profile_completion }, next_url: "/member/dashboard/" }); } catch (error) { console.error( "Member login error:", error ); return sendError( res, 500, "MEMBER_LOGIN_FAILED", "ไม่สามารถเข้าสู่ระบบได้ กรุณาลองใหม่อีกครั้ง" ); } } ); return router; };
